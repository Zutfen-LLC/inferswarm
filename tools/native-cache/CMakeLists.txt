cmake_minimum_required(VERSION 3.14)
project(inferswarm_rpc_cache_probe C CXX ASM)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(GGML_RPC ON CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(../native-source/ggml ggml)
add_executable(ggml-rpc-server ../native-source/tools/rpc/rpc-server.cpp)
target_link_libraries(ggml-rpc-server PRIVATE ggml)
target_compile_features(ggml-rpc-server PRIVATE cxx_std_17)
add_executable(driver driver.cpp)
target_link_libraries(driver PRIVATE ggml ggml-rpc)
target_compile_features(driver PRIVATE cxx_std_17)
